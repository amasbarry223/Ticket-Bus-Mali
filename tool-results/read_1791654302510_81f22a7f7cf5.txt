    73→/** Normalise une chaîne pour la recherche : minuscules, sans espaces/tirets/points. */
    74→function normalizeKey(s: string): string {
    75→  return s
    76→    .toLowerCase()
    77→    .replace(/[\s\-.]/g, "")
    78→    .normalize("NFD")
    79→    .replace(/[\u0300-\u036f]/g, "");
    80→}
    81→
    82→function initials(name: string): string {
    83→  return name
    84→    .split(/\s+/)
    85→    .map((p) => p[0] ?? "")
    86→    .join("")
    87→    .slice(0, 2)
    88→    .toUpperCase();
    89→}
