    73→/** Initiales pour l'avatar-logo d'une compagnie (ex. "Bamako Express" → BE). */
    74→export function companyInitials(nom: string): string {
    75→  return (
    76→    nom
    77→      .split(/\s+/)
    78→      .map((p) => p[0])
    79→      .slice(0, 2)
    80→      .join("")
    81→      .toUpperCase() || "CP"
    82→  );
    83→}

  1003→/** Montant déjà remboursé (somme des versements opérateur). */
  1004→export function refundMontantRembourse(r: Refund): number {
  1005→  return r.versements.reduce((s, v) => s + v.montant, 0);
  1006→}
  1007→
  1008→/** Reste à rembourser sur le dossier. */
  1009→export function refundResteDu(r: Refund): number {
  1010→  return Math.max(0, r.montantRemboursable - refundMontantRembourse(r));
  1011→}
