   145→/**
   146→ * État opérationnel du back-office (frontend only).
   147→ * Chaque action sensible alimente automatiquement le journal d'audit (A06).
   148→ */
   149→export const useTbmStore = create<TbmState>()((set) => ({
   150→  companies: initialCompanies,
   151→  users: initialUserAccounts,
   152→  refunds: initialRefunds,
   153→  audit: initialAudit,
   154→  settings: defaultSettings,
   155→  transactions: initialTransactions,
   156→  closedReconciliations: [],
   157→  notifications: initialNotifications,
   158→  settlements: initialSettlements,
   159→
   160→  setCompanyStatus: (id, statut, auteur) =>
   161→    set((state) => {
   162→      const company = state.companies.find((c) => c.id === id);
   163→      if (!company) return state;
   164→      const entry = makeAudit({
   165→        date: "À l'instant",
   166→        auteur,
   167→        role: "Session en cours",
   168→        action: "Compagnie" as AuditAction,
   169→        objet: company.nom,
   170→        detail:
   171→          statut === "active"
   172→            ? "Compagnie activée — vente en ligne autorisée."
   173→            : statut === "suspendue"
   174→              ? "Compagnie suspendue — ventes bloquées."
   175→              : "Compagnie remise en attente de validation.",
   176→      });
   177→      return {
   178→        companies: state.companies.map((c) =>
   179→          c.id === id ? { ...c, statut } : c
   180→        ),
   181→        audit: [entry, ...state.audit],
   182→      };
   183→    }),
   184→
   185→  addCompany: (company, auteur) =>
   186→    set((state) => ({
   187→      companies: [company, ...state.companies],
   188→      audit: [
   189→        makeAudit({
   190→          date: "À l'instant",
   191→          auteur,
   192→          role: "Session en cours",
   193→          action: "Compagnie",
   194→          objet: company.nom,
   195→          detail: `Nouvelle compagnie créée (commission ${company.commission} %, ${company.ville}).`,
   196→        }),
   197→        ...state.audit,
   198→      ],
   199→    })),
   200→
   201→  updateCompany: (id, patch, auteur) =>
   202→    set((state) => {
   203→      const company = state.companies.find((c) => c.id === id);
   204→      if (!company) return state;
   205→      const changes: string[] = [];
   206→      if (patch.nom && patch.nom !== company.nom)
   207→        changes.push(`nom → « ${patch.nom} »`);
   208→      if (patch.contact && patch.contact !== company.contact)
   209→        changes.push(`e-mail → ${patch.contact}`);
   210→      if (patch.telephone && patch.telephone !== company.telephone)
   211→        changes.push(`téléphone → ${patch.telephone}`);
   212→      if (patch.adresse && patch.adresse !== company.adresse)
   213→        changes.push(`adresse → ${patch.adresse}`);
   214→      if (patch.ville && patch.ville !== company.ville)
   215→        changes.push(`ville → ${patch.ville}`);
   216→      if (patch.commission !== undefined && patch.commission !== company.commission)
   217→        changes.push(`commission ${company.commission} % → ${patch.commission} %`);
   218→      if (patch.params) {
   219→        const p = patch.params;
   220→        const o = company.params;
   221→        if (p.venteEnLigne !== o.venteEnLigne)
   222→          changes.push(`vente en ligne ${p.venteEnLigne ? "activée" : "désactivée"}`);
   223→        if (p.venteGuichet !== o.venteGuichet)
   224→          changes.push(`vente au guichet ${p.venteGuichet ? "activée" : "désactivée"}`);
   225→        if (p.reservationJours !== o.reservationJours)
   226→          changes.push(`réservation ${o.reservationJours} → ${p.reservationJours} j`);
   227→        if (p.annulationGratuite !== o.annulationGratuite)
   228→          changes.push(`annulation gratuite ${o.annulationGratuite} → ${p.annulationGratuite} h`);
   229→      }
   230→      return {
   231→        companies: state.companies.map((c) =>
   232→          c.id === id ? { ...c, ...patch } : c
   233→        ),
   234→        audit: [
   235→          makeAudit({
   236→            date: "À l'instant",
   237→            auteur,
   238→            role: "Session en cours",
   239→            action: "Compagnie" as AuditAction,
   240→            objet: company.nom,
   241→            detail:
   242→              changes.length > 0
   243→                ? `Fiche modifiée : ${changes.join(" · ")}.`
   244→                : "Fiche enregistrée sans modification.",
   245→          }),
   246→          ...state.audit,
   247→        ],
   248→      };
   249→    }),
   250→
   251→  /**
   252→   * Création d'un compte professionnel — le compte reste « À activer »
   253→   * jusqu'à validation TBM (aucun accès tant que le compte n'est pas activé).
   254→   */
   255→  createProAccount: (input, auteur, role) =>
   256→    set((state) => {
   257→      const account: UserAccount = {
   258→        id: `USR-${userSeq++}`,
   259→        nom: input.nom.trim(),
   260→        email: input.email.trim().toLowerCase(),
   261→        telephone: input.telephone.trim() || undefined,
   262→        role: input.role,
   263→        compagnie: input.compagnie,
   264→        statut: "attente",
   265→        creeLe: "À l'instant",
   266→        creePar: auteur,
   267→        connexions: [],
   268→        actionsSensibles: [],
   269→      };
   270→      return {
   271→        users: [account, ...state.users],
   272→        audit: [
   273→          makeAudit({
   274→            date: "À l'instant",
   275→            auteur,
   276→            role,
   277→            action: "Utilisateur" as AuditAction,
   278→            objet: account.nom,
   279→            detail: `Compte professionnel créé (${account.role} · ${account.compagnie}) — en attente d'activation, aucun accès tant que le compte n'est pas activé.`,
   280→          }),
   281→          ...state.audit,
   282→        ],
   283→      };
   284→    }),
   285→
   286→  /** Activation d'un compte professionnel — mot de passe provisoire remis par SMS. */
   287→  activateUser: (id, auteur, role) =>
   288→    set((state) => {
   289→      const user = state.users.find((u) => u.id === id);
   290→      if (!user || user.statut !== "attente") return state;
   291→      return {
   292→        users: state.users.map((u) =>
   293→          u.id === id
   294→            ? {
   295→                ...u,
   296→                statut: "actif" as const,
   297→                activeLe: "À l'instant",
   298→                activePar: auteur,
   299→              }
   300→            : u
   301→        ),
   302→        audit: [
   303→          makeAudit({
   304→            date: "À l'instant",
   305→            auteur,
   306→            role,
   307→            action: "Utilisateur" as AuditAction,
   308→            objet: user.nom,
   309→            detail: `Compte professionnel activé (${user.role}${user.compagnie ? ` · ${user.compagnie}` : ""}) — mot de passe provisoire envoyé par SMS.`,
   310→          }),
   311→          ...state.audit,
   312→        ],
   313→      };
   314→    }),
   315→
   316→  /**
   317→   * Blocage d'un compte — motif obligatoire et tracé.
   318→   * Un administrateur TBM ne peut jamais être bloqué (protection RBAC).
   319→   */
   320→  blockUser: (id, motif, auteur, role) =>
   321→    set((state) => {
   322→      const user = state.users.find((u) => u.id === id);
   323→      if (
   324→        !user ||
   325→        user.statut === "bloque" ||
   326→        user.role === "Administrateur TBM" ||
   327→        !motif.trim()
   328→      )
   329→        return state;
   330→      return {
   331→        users: state.users.map((u) =>
   332→          u.id === id
   333→            ? {
   334→                ...u,
   335→                statut: "bloque" as const,
   336→                bloqueMotif: motif.trim(),
   337→                bloquePar: auteur,
   338→                bloqueLe: "À l'instant",
   339→              }
   340→            : u
   341→        ),
   342→        audit: [
   343→          makeAudit({
   344→            date: "À l'instant",
   345→            auteur,
   346→            role,
   347→            action: "Utilisateur" as AuditAction,
   348→            objet: user.nom,
   349→            detail: `Compte bloqué — « ${motif.trim()} » — accès back-office et application retirés.`,
   350→          }),
   351→          ...state.audit,
   352→        ],
   353→      };
   354→    }),
   355→
   356→  /** Déblocage d'un compte — après vérification de l'identité du titulaire. */
   357→  unblockUser: (id, auteur, role) =>
   358→    set((state) => {
   359→      const user = state.users.find((u) => u.id === id);
   360→      if (!user || user.statut !== "bloque") return state;
   361→      return {
   362→        users: state.users.map((u) =>
   363→          u.id === id
   364→            ? {
   365→                ...u,
   366→                statut: "actif" as const,
   367→                bloqueMotif: undefined,
   368→                bloquePar: undefined,
   369→                bloqueLe: undefined,
   370→              }
   371→            : u
   372→        ),
   373→        audit: [
   374→          makeAudit({
   375→            date: "À l'instant",
   376→            auteur,
   377→            role,
   378→            action: "Utilisateur" as AuditAction,
   379→            objet: user.nom,
   380→            detail: "Compte débloqué — accès rétabli après vérification de l'identité.",
   381→          }),
   382→          ...state.audit,
   383→        ],
   384→      };
   385→    }),
   386→
   387→  /**
   388→   * Changement de rôle — justification obligatoire, tracée au journal d'audit.
   389→   * Le périmètre d'isolation est recalculé automatiquement à partir du rôle.
   390→   */
   391→  changeUserRole: (id, nouveauRole, justification, auteur, role) =>
   392→    set((state) => {
   393→      const user = state.users.find((u) => u.id === id);
   394→      if (
   395→        !user ||
   396→        user.role === nouveauRole ||
   397→        user.role === "Administrateur TBM" ||
   398→        !justification.trim()
   399→      )
   400→        return state;
   401→      return {
   402→        users: state.users.map((u) =>
   403→          u.id === id ? { ...u, role: nouveauRole } : u
   404→        ),
   405→        audit: [
   406→          makeAudit({
   407→            date: "À l'instant",
   408→            auteur,
   409→            role,
   410→            action: "Utilisateur" as AuditAction,
   411→            objet: user.nom,
   412→            detail: `Rôle modifié : ${user.role} → ${nouveauRole} — « ${justification.trim()} ». Périmètre d'accès recalculé.`,
   413→          }),
   414→          ...state.audit,
   415→        ],
   416→      };
   417→    }),
   418→
   419→  /** Prise en charge d'un dossier : l'agent devient responsable du traitement. */
   420→  claimRefund: (id, auteur, role) =>
   421→    set((state) => {
   422→      const refund = state.refunds.find((r) => r.id === id);
   423→      if (!refund || refund.traitePar) return state;
   424→      const event: RefundEvent = {
   425→        id: `RVE-${refundEventSeq++}`,
   426→        date: "À l'instant",
   427→        type: "prise_en_charge",
   428→        detail:
   429→          "Dossier pris en charge — vérification du motif, du débit d'origine et des pièces jointes.",
   430→        par: auteur,
   431→        role,
   432→      };
   433→      return {
   434→        refunds: state.refunds.map((r) =>
   435→          r.id === id
   436→            ? { ...r, traitePar: auteur, historique: [...r.historique, event] }
   437→            : r
   438→        ),
   439→        audit: [
   440→          makeAudit({
   441→            date: "À l'instant",
   442→            auteur,
   443→            role: "Session en cours",
   444→            action: "Remboursement" as AuditAction,
   445→            objet: `${refund.id} · ${refund.passager}`,
   446→            detail: "Dossier de remboursement pris en charge.",
   447→          }),
   448→          ...state.audit,
   449→        ],
   450→      };
   451→    }),
   452→
   453→  /**
   454→   * Décision manuelle sur un dossier — justification obligatoire et tracée.
   455→   * Un dossier en litige est gelé jusqu'à la levée du litige (droits admin).
   456→   */
   457→  decideRefund: (id, approuve, justification, auteur, role) =>
   458→    set((state) => {
   459→      const refund = state.refunds.find((r) => r.id === id);
   460→      if (!refund || refund.statut !== "en_attente" || refund.litige?.ouvert)
   461→        return state;
   462→      const event: RefundEvent = {
   463→        id: `RVE-${refundEventSeq++}`,
   464→        date: "À l'instant",
   465→        type: "decision",
   466→        detail: approuve
   467→          ? `Remboursement validé — « ${justification} »`
   468→          : `Remboursement refusé — « ${justification} »`,
   469→        par: auteur,
   470→        role,
   471→      };
   472→      return {
   473→        refunds: state.refunds.map((r) =>
   474→          r.id === id
   475→            ? {
   476→                ...r,
   477→                statut: approuve ? ("approuve" as const) : ("refuse" as const),
   478→                traitePar: auteur,
   479→                justification,
   480→                historique: [...r.historique, event],
   481→              }
   482→            : r
   483→        ),
   484→        audit: [
   485→          makeAudit({
   486→            date: "À l'instant",
   487→            auteur,
   488→            role: "Session en cours",
   489→            action: "Remboursement" as AuditAction,
   490→            objet: `${refund.id} · ${refund.passager}`,
   491→            detail: approuve
   492→              ? `Remboursement de ${new Intl.NumberFormat("fr-FR").format(refund.montantRemboursable)} FCFA validé — « ${justification} »`
   493→              : `Remboursement refusé — « ${justification} »`,
   494→          }),
   495→          ...state.audit,
   496→        ],
   497→      };
   498→    }),
   499→
   500→  /**
   501→   * Enregistrement d'un versement opérateur (tranche de remboursement).
   502→   * Le dossier passe à « Remboursé » lorsque le montant éligible est couvert.
   503→   */
   504→  addRefundVersement: (id, montant, refOperateur, auteur, role) =>
   505→    set((state) => {
   506→      const refund = state.refunds.find((r) => r.id === id);
   507→      if (
   508→        !refund ||
   509→        refund.statut !== "approuve" ||
   510→        refund.litige?.ouvert ||
   511→        montant <= 0 ||
   512→        !refOperateur.trim()
   513→      )
   514→        return state;
   515→      const dejaRembourse = refundMontantRembourse(refund);
   516→      const resteAvant = Math.max(
   517→        0,
   518→        refund.montantRemboursable - dejaRembourse
   519→      );
   520→      const montantFinal = Math.min(montant, resteAvant);
   521→      const solde = montantFinal >= resteAvant;
   522→      const rang = refund.versements.length + 1;
   523→      const versement: RefundVersement = {
   524→        id: `RVS-${refundVersementSeq++}`,
   525→        date: "À l'instant",
   526→        montant: montantFinal,
   527→        refOperateur: refOperateur.trim(),
   528→        par: auteur,
   529→      };
   530→      const refClean = refOperateur.trim();
   531→      const montantTxt = new Intl.NumberFormat("fr-FR").format(montantFinal);
   532→      const resteTxt = new Intl.NumberFormat("fr-FR").format(
   533→        resteAvant - montantFinal
   534→      );
   535→      const event: RefundEvent = {
   536→        id: `RVE-${refundEventSeq++}`,
   537→        date: "À l'instant",
   538→        type: "versement",
   539→        detail: solde
   540→          ? `Versement de ${montantTxt} FCFA exécuté (réf. ${refClean}) — dossier intégralement remboursé.`
   541→          : `${rang === 1 ? "1er versement" : `Versement n°${rang}`} de ${montantTxt} FCFA exécuté (réf. ${refClean}) — reste ${resteTxt} FCFA à verser.`,
   542→        par: auteur,
   543→        role,
   544→      };
   545→      return {
   546→        refunds: state.refunds.map((r) =>
   547→          r.id === id
   548→            ? {
   549→                ...r,
   550→                statut: solde ? ("rembourse" as const) : r.statut,
   551→                versements: [...r.versements, versement],
   552→                historique: [...r.historique, event],
   553→              }
   554→            : r
   555→        ),
   556→        audit: [
   557→          makeAudit({
   558→            date: "À l'instant",
   559→            auteur,
   560→            role: "Session en cours",
   561→            action: "Remboursement" as AuditAction,
   562→            objet: `${refund.id} · ${refund.passager}`,
   563→            detail: `Versement opérateur de ${montantTxt} FCFA enregistré (réf. ${refClean})${solde ? " — dossier soldé" : ""}.`,
   564→          }),
   565→          ...state.audit,
   566→        ],
   567→      };
   568→    }),
   569→
   570→  /** Escalade d'un dossier en litige — gel des décisions jusqu'à résolution. */
   571→  openRefundDispute: (id, detail, auteur, role) =>
   572→    set((state) => {
   573→      const refund = state.refunds.find((r) => r.id === id);
   574→      if (!refund || refund.litige?.ouvert) return state;
   575→      const event: RefundEvent = {
   576→        id: `RVE-${refundEventSeq++}`,
   577→        date: "À l'instant",
