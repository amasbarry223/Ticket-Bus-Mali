    71→/** Normalise une chaîne pour la recherche : minuscules, sans espaces/tirets/points. */
    72→function normalizeKey(s: string): string {
    73→  return s.toLowerCase().replace(/[\s.\-+()]/g, "");
    74→}
    75→
    76→/** Empreinte de recherche d'une réservation : référence, téléphone, billets, passagers. */
    77→function searchKey(r: Reservation): string {
    78→  return normalizeKey(
    79→    [
    80→      r.id,
    81→      r.telephone,
    82→      ...r.tickets.map((t) => t.id),
    83→      ...r.tickets.map((t) => t.passager.nom),
    84→    ].join(" ")
    85→  );
    86→}
    87→
    88→function tarifsTotal(r: Reservation): number {
    89→  return r.tickets.reduce((a, t) => a + t.tarif, 0);
    90→}

    92→/** Frais de service facturés sur la réservation (paramétrage A04). */
    93→function serviceFees(r: Reservation, settings: Settings): number {
    94→  if (r.tickets.length === 0) return 0;
    95→  return settings.fraisMode === "fixe"
    96→    ? settings.fraisValeur * r.tickets.length
    97→    : Math.round((tarifsTotal(r) * settings.fraisValeur) / 100);
    98→}
    99→
   100→function reservationTotal(r: Reservation, settings: Settings): number {
   101→  return tarifsTotal(r) + serviceFees(r, settings);
   102→}

   272→/** Grille pseudo-QR déterministe (démo frontend) — pure, calculée hors rendu. */
   273→function buildQrGrid(seed: string): boolean[][] {
   274→  const n = 21;
   275→  let h = 2166136261;
   276→  for (let i = 0; i < seed.length; i++) {
   277→    h ^= seed.charCodeAt(i);
   278→    h = Math.imul(h, 16777619);
   279→  }
   280→  const rand = () => {
   281→    h ^= h << 13;
   282→    h ^= h >>> 17;
   283→    h ^= h << 5;
   284→    return (h >>> 0) / 4294967296;
   285→  };
   286→  const grid: boolean[][] = Array.from({ length: n }, () =>
   287→    Array.from({ length: n }, () => rand() > 0.52)
   288→  );
   289→  const finder = (ox: number, oy: number) => {
   290→    for (let y = -1; y <= 7; y++) {
   291→      for (let x = -1; x <= 7; x++) {
   292→        const gx = ox + x;
   293→        const gy = oy + y;
   294→        if (gx < 0 || gy < 0 || gx >= n || gy >= n) continue;
   295→        const edge = x === 0 || x === 6 || y === 0 || y === 6;
   296→        const core = x >= 2 && x <= 4 && y >= 2 && y <= 4;
   297→        grid[gy][gx] =
   298→          x === -1 || x === 7 || y === -1 || y === 7 ? false : edge || core;
   299→      }
   300→    }
   301→  };
   302→  finder(0, 0);
   303→  finder(14, 0);
   304→  finder(0, 14);
   305→  return grid;
   306→}

   308→/** Pseudo-QR déterministe (démo frontend) — grille 21×21 avec motifs de repérage. */
   309→function QrCodeSvg({
   310→  seed,
   311→  className,
   312→}: {

   315→}) {

   324→    >
   325→      <rect width="21" height="21" fill="#ffffff" />
   326→      {cells.map((row, y) =>
   327→        row.map((on, x) =>
   328→          on ? (

  1213→/** Ajoute des minutes à une heure "HH:MM" (heure estimée après retard). */
  1214→function shiftTime(heure: string, minutes: number): string {
  1215→  const [h, m] = heure.split(":").map(Number);
  1216→  const total = h * 60 + m + minutes;
  1217→  const hh = Math.floor(total / 60) % 24;
  1218→  const mm = total % 60;
  1219→  return `${String(hh).padStart(2, "0")}:${String(mm).padStart(2, "0")}`;
  1220→}
