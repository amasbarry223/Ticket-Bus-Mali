    74→    74→export function companyInitials(nom: string): string {
    75→    75→  return (
    76→    76→    nom
    77→    77→      .split(/\s+/)
    78→    78→      .map((p) => p[0])
    79→    79→      .slice(0, 2)
    80→    80→      .join("")
    81→    81→      .toUpperCase() || "CP"
    82→    82→  );
    83→    83→}
