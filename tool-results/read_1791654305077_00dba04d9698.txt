    84→/** Normalise une chaîne pour la recherche : minuscules, sans espaces/tirets/points. */
    85→function normalizeKey(s: string): string {
    86→  return s
    87→    .toLowerCase()
    88→    .replace(/[\s\-.]/g, "")
    89→    .normalize("NFD")
    90→    .replace(/[\u0300-\u036f]/g, "");
    91→}

   266→function CopyRef({ value }: { value: string }) {
   267→  const [copied, setCopied] = useState(false);
   268→  const { toast } = useToast();
   269→  return (
   270→    <button
   271→      type="button"
   272→      aria-label={`Copier la référence ${value}`}
   273→      className="inline-flex h-6 w-6 items-center justify-center rounded-md text-neutral-400 transition-colors hover:bg-neutral-100 hover:text-neutral-600"
   274→      onClick={() => {
   275→        navigator.clipboard?.writeText(value).catch(() => undefined);
   276→        setCopied(true);
   277→        toast({ title: "Référence copiée", description: value });
   278→        window.setTimeout(() => setCopied(false), 1600);
   279→      }}
   280→    >
   281→      {copied ? (
   282→        <Check className="h-3.5 w-3.5 text-emerald-600" />
   283→      ) : (
   284→        <Copy className="h-3.5 w-3.5" />
   285→      )}
   286→    </button>
   287→  );
   288→}
