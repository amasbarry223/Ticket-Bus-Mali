   287→   287→ * Consultation des lignes interrégionales (arrêts, tarifs, capacité),
   288→   288→ * tableau des départs avec statuts et suivi des modifications de programme.

   291→   291→export function LinesPage() {
   292→   292→  const companies = useTbmStore((s) => s.companies);
   293→   293→
   294→   294→  const [view, setView] = useState<"departs" | "lignes">("departs");
   295→   295→  const [query, setQuery] = useState("");
   296→   296→  const [companyFilter, setCompanyFilter] = useState("all");
   297→   297→  const [statusFilter, setStatusFilter] = useState("all");
   298→   298→  const [dayFilter, setDayFilter] = useState("all");
   299→   299→  const [selectedDepId, setSelectedDepId] = useState<string | null>(null);
   300→   300→  const [selectedLineKey, setSelectedLineKey] = useState<string | null>(null);
   301→   301→
   302→   302→  const companyById = useMemo(
   303→   303→    () => new Map(companies.map((c) => [c.id, c])),
   304→   304→    [companies]
   305→   305→  );
   306→   306→
   307→   307→  /* ---- KPIs du jour ---- */
   308→   308→  const kpis = useMemo(() => {
   309→   309→    const today = companyDepartures.filter((d) => d.jour === "Aujourd'hui");
   310→   310→    const sold = today.reduce((a, d) => a + (d.placesTotal - d.placesDispo), 0);
   311→   311→    const cap = today.reduce((a, d) => a + d.placesTotal, 0);
   312→   312→    const occ = cap > 0 ? Math.round((sold / cap) * 100) : 0;
   313→   313→    const perturbes = today.filter(
   314→   314→      (d) => d.statut === "retarde" || d.statut === "annule"
   315→   315→    ).length;
   316→   316→    return { count: today.length, sold, occ, perturbes, modifs: departureModifications.length };
   317→   317→  }, []);
   318→   318→
   319→   319→  /* ---- Tableau des départs filtré ---- */
   320→   320→  const filteredDeparts = useMemo(() => {
   321→   321→    const q = query.trim().toLowerCase();
   322→   322→    const dayRank: Record<string, number> = {
   323→   323→      "Aujourd'hui": 0,
   324→   324→      Demain: 1,
   325→   325→    };
   326→   326→    return companyDepartures
   327→   327→      .filter((d) => {
   328→   328→        const company = companyById.get(d.compagnie);
   329→   329→        const matchQuery =
   330→   330→          q === "" ||
   331→   331→          d.trajet.toLowerCase().includes(q) ||
   332→   332→          d.bus.toLowerCase().includes(q) ||
   333→   333→          (company?.nom.toLowerCase().includes(q) ?? false);
   334→   334→        const matchCompany =
   335→   335→          companyFilter === "all" || d.compagnie === companyFilter;
   336→   336→        const matchStatus =
   337→   337→          statusFilter === "all" || d.statut === (statusFilter as DepartureStatus);
   338→   338→        const matchDay = dayFilter === "all" || d.jour === dayFilter;
   339→   339→        return matchQuery && matchCompany && matchStatus && matchDay;
   340→   340→      })
   341→   341→      .sort((a, b) => {
   342→   342→        const jr = (dayRank[a.jour] ?? 2) - (dayRank[b.jour] ?? 2);
   343→   343→        if (jr !== 0) return jr;
   344→   344→        return a.heure.localeCompare(b.heure);
   345→   345→      });
   346→   346→  }, [query, companyFilter, statusFilter, dayFilter, companyById]);

   371→   371→        duree: l.duree,

   377→   377→        compagnieId: l.compagnie,
   378→   378→        prix: l.prix,
   379→   379→        frequence: l.frequence,
   380→   380→      });
   381→   381→      map.set(key, cur);
   382→   382→    }

   409→   409→    [selectedDep]

   411→   411→  const depStops = useMemo(() => {

   418→   418→  }, [selectedDep]);
   419→   419→
   420→   420→  /* ---- Fiche détail ligne ---- */
   421→   421→  const selectedLine = network.find((n) => n.key === selectedLineKey) ?? null;
   422→   422→  const lineDepartures = useMemo(
   423→   423→    () =>
   424→   424→      selectedLine
   425→   425→        ? companyDepartures
   426→   426→            .filter((d) => d.trajet === selectedLine.key && d.statut !== "termine")
   427→   427→            .sort((a, b) => a.heure.localeCompare(b.heure))
   428→   428→        : [],
   429→   429→    [selectedLine]
   430→   430→  );
   431→   431→
   432→   432→  return (
   433→   433→    <div>
   434→   434→      {/* ---- En-tête ---- */}
   435→   435→      <div className="flex flex-wrap items-start justify-between gap-3">
   436→   436→        <div>
   437→   437→          <h2 className="text-[17px] font-bold tracking-tight text-neutral-900">
   438→   438→            Lignes &amp; Départs
   439→   439→          </h2>
   440→   440→          <p className="mt-0.5 max-w-xl text-[12.5px] text-neutral-400">
   441→   441→            A09 — Supervision des départs de toutes les compagnies : lecture
   442→   442→            seule, la programmation reste gérée par l’outil de chaque compagnie.
   443→   443→          </p>
   444→   444→        </div>
   445→   445→      </div>
   446→   446→
   447→   447→      {/* ---- Indicateurs du jour ---- */}
   448→   448→      <div className="mt-4 grid grid-cols-2 gap-3 xl:grid-cols-4">
   449→   449→        <Kpi
   450→   450→          index={0}
   451→   451→          label="Départs aujourd’hui"
   452→   452→          value={formatNumber(kpis.count)}
   453→   453→          hint={`${formatNumber(companyDepartures.length)} sur 2 jours`}

  1141→  1141→/** Ajoute des minutes à une heure "HH:MM" (utilisée pour l'heure estimée). */

  1143→  1143→  const [h, m] = heure.split(":").map(Number);
  1144→  1144→  const total = h * 60 + m + minutes;
  1145→  1145→  const hh = Math.floor(total / 60) % 24;
  1146→  1146→  const mm = total % 60;
  1147→  1147→  return `${String(hh).padStart(2, "0")}:${String(mm).padStart(2, "0")}`;
  1148→  1148→}
