    56→function Kpi({
    57→  label,
    58→  value,
    59→  hint,
    60→  icon: Icon,
    61→  tone,
    62→  index,
    63→}: {
    64→  label: string;
    65→  value: string;
    66→  hint: string;
    67→  icon: React.ComponentType<{ className?: string }>;
    68→  tone: "coral" | "amber" | "rose" | "neutral";
    69→  index: number;
    70→}) {
    71→  const tones: Record<string, string> = {
    72→    coral: "group-hover:bg-coral-50 group-hover:text-coral-500",
    73→    amber: "group-hover:bg-amber-50 group-hover:text-amber-600",
    74→    rose: "group-hover:bg-rose-50 group-hover:text-rose-600",
    75→    neutral: "group-hover:bg-neutral-200/60",
    76→  };
    77→  return (
    78→    <motion.article
    79→      initial={{ opacity: 0, y: 14 }}
    80→      animate={{ opacity: 1, y: 0 }}
    81→      transition={{
    82→        duration: 0.45,
    83→        delay: 0.05 + index * 0.06,
    84→        ease: [0.22, 1, 0.36, 1],
    85→      }}
    86→      className="group ff-card rounded-2xl border border-neutral-200/70 bg-white p-4"
    87→    >
    88→      <div className="flex items-start justify-between gap-2">
    89→        <h3 className="text-[12.5px] font-medium text-neutral-500">{label}</h3>
    90→        <span
    91→          className={cn(
    92→            "flex h-8 w-8 items-center justify-center rounded-xl bg-neutral-100 text-neutral-500 transition-colors duration-300",
    93→            tones[tone]
    94→          )}
    95→        >
    96→          <Icon className="h-4 w-4" aria-hidden="true" />
    97→        </span>
    98→      </div>
    99→      <p className="mt-2.5 text-[21px] font-bold leading-none tracking-tight text-neutral-900">
   100→        {value}
   101→      </p>
   102→      <p className="mt-1.5 text-[11.5px] text-neutral-400">{hint}</p>
   103→    </motion.article>
   104→  );
   105→}

   123→function OccupancyBar({
   124→  total,
   125→  dispo,
   126→  muted,
   127→}: {
   128→  total: number;

   140→            muted ? "bg-neutral-200" : pct >= 80 ? "bg-coral-500" : "bg-coral-400/70"
   141→          )}
   142→          style={{ width: `${muted ? 0 : pct}%` }}
   143→        />
   144→      </div>

   189→          {i < points.length - 1 && (
   190→            <span

   192→              className="absolute left-[5px] top-3 h-full w-px bg-neutral-200"
   193→            />

   285→/**
   286→ * A09 — Lignes & Départs : supervision inter-compagnies (lecture seule).
   287→ * Consultation des lignes interrégionales (arrêts, tarifs, capacité),
   288→ * tableau des départs avec statuts et suivi des modifications de programme.
   289→ * Ne remplace pas l'outil de programmation propre à chaque compagnie.
   290→ */
   291→export function LinesPage() {
   292→  const companies = useTbmStore((s) => s.companies);
   293→
   294→  const [view, setView] = useState<"departs" | "lignes">("departs");
   295→  const [query, setQuery] = useState("");
   296→  const [companyFilter, setCompanyFilter] = useState("all");
   297→  const [statusFilter, setStatusFilter] = useState("all");
   298→  const [dayFilter, setDayFilter] = useState("all");
   299→  const [selectedDepId, setSelectedDepId] = useState<string | null>(null);
   300→  const [selectedLineKey, setSelectedLineKey] = useState<string | null>(null);
   301→
   302→  const companyById = useMemo(
   303→    () => new Map(companies.map((c) => [c.id, c])),
   304→    [companies]
   305→  );
   306→
   307→  /* ---- KPIs du jour ---- */
   308→  const kpis = useMemo(() => {
   309→    const today = companyDepartures.filter((d) => d.jour === "Aujourd'hui");
   310→    const sold = today.reduce((a, d) => a + (d.placesTotal - d.placesDispo), 0);
   311→    const cap = today.reduce((a, d) => a + d.placesTotal, 0);
   312→    const occ = cap > 0 ? Math.round((sold / cap) * 100) : 0;
   313→    const perturbes = today.filter(
   314→      (d) => d.statut === "retarde" || d.statut === "annule"
   315→    ).length;
   316→    return { count: today.length, sold, occ, perturbes, modifs: departureModifications.length };
   317→  }, []);
   318→
   319→  /* ---- Tableau des départs filtré ---- */
   320→  const filteredDeparts = useMemo(() => {
   321→    const q = query.trim().toLowerCase();
   322→    const dayRank: Record<string, number> = {
   323→      "Aujourd'hui": 0,
   324→      Demain: 1,
   325→    };
   326→    return companyDepartures
   327→      .filter((d) => {
   328→        const company = companyById.get(d.compagnie);
   329→        const matchQuery =
   330→          q === "" ||
   331→          d.trajet.toLowerCase().includes(q) ||
   332→          d.bus.toLowerCase().includes(q) ||
   333→          (company?.nom.toLowerCase().includes(q) ?? false);
   334→        const matchCompany =
   335→          companyFilter === "all" || d.compagnie === companyFilter;
   336→        const matchStatus =
   337→          statusFilter === "all" || d.statut === (statusFilter as DepartureStatus);
   338→        const matchDay = dayFilter === "all" || d.jour === dayFilter;
   339→        return matchQuery && matchCompany && matchStatus && matchDay;
   340→      })
   341→      .sort((a, b) => {
   342→        const jr = (dayRank[a.jour] ?? 2) - (dayRank[b.jour] ?? 2);
   343→        if (jr !== 0) return jr;
   344→        return a.heure.localeCompare(b.heure);
   345→      });
   346→  }, [query, companyFilter, statusFilter, dayFilter, companyById]);
   347→
   348→  /* ---- Réseau de lignes (agrégé inter-compagnies) ---- */
   349→  interface Offering {
   350→    compagnieId: string;
   351→    prix: number;
   352→    frequence: string;
   353→  }
   354→  interface NetworkLine {
   355→    key: string;
   356→    depart: string;
   357→    arrivee: string;
   358→    duree: string;
   359→    distanceKm: number;
   360→    arrets: string[];
   361→    offerings: Offering[];
   362→  }
   363→  const network = useMemo(() => {
   364→    const map = new Map<string, NetworkLine>();
   365→    for (const l of companyLines) {
   366→      const key = `${l.depart} → ${l.arrivee}`;
   367→      const cur = map.get(key) ?? {
   368→        key,
   369→        depart: l.depart,
   370→        arrivee: l.arrivee,
   371→        duree: l.duree,
   372→        distanceKm: l.distanceKm,
   373→        arrets: l.arrets,
   374→        offerings: [],
   375→      };
   376→      cur.offerings.push({
   377→        compagnieId: l.compagnie,
   378→        prix: l.prix,
   379→        frequence: l.frequence,
   380→      });
   381→      map.set(key, cur);
   382→    }
   383→    return [...map.values()];
   384→  }, []);
   385→
   386→  const filteredNetwork = useMemo(() => {
   387→    const q = query.trim().toLowerCase();
   388→    if (q === "") return network;
   389→    return network.filter(
   390→      (n) =>
   391→        n.key.toLowerCase().includes(q) ||
   392→        n.arrets.some((a) => a.toLowerCase().includes(q)) ||
   393→        n.offerings.some((o) =>
   394→          (companyById.get(o.compagnieId)?.nom ?? "")
   395→            .toLowerCase()
   396→            .includes(q)
   397→        )
   398→    );
   399→  }, [network, query, companyById]);
   400→
   401→  /* ---- Fiche détail départ ---- */
   402→  const selectedDep = companyDepartures.find((d) => d.id === selectedDepId) ?? null;
   403→  const depCompany = selectedDep ? companyById.get(selectedDep.compagnie) : null;
   404→  const depMods = useMemo(
   405→    () =>
   406→      selectedDep
   407→        ? departureModifications.filter((m) => m.departId === selectedDep.id)
   408→        : [],
   409→    [selectedDep]
   410→  );
   411→  const depStops = useMemo(() => {
   412→    if (!selectedDep) return null;
   413→    const [depart, arrivee] = selectedDep.trajet.split(" → ");
   414→    const line = companyLines.find(
   415→      (l) => l.depart === depart && l.arrivee === arrivee
   416→    );
   417→    return line ? { depart, arrivee, arrets: line.arrets } : { depart, arrivee, arrets: [] };
   418→  }, [selectedDep]);
   419→
   420→  /* ---- Fiche détail ligne ---- */
   421→  const selectedLine = network.find((n) => n.key === selectedLineKey) ?? null;
   422→  const lineDepartures = useMemo(
   423→    () =>
   424→      selectedLine
   425→        ? companyDepartures
   426→            .filter((d) => d.trajet === selectedLine.key && d.statut !== "termine")
   427→            .sort((a, b) => a.heure.localeCompare(b.heure))
   428→        : [],
   429→    [selectedLine]
   430→  );
   431→
   432→  return (
   433→    <div>
   434→      {/* ---- En-tête ---- */}
   435→      <div className="flex flex-wrap items-start justify-between gap-3">
   436→        <div>
   437→          <h2 className="text-[17px] font-bold tracking-tight text-neutral-900">
   438→            Lignes &amp; Départs
   439→          </h2>
   440→          <p className="mt-0.5 max-w-xl text-[12.5px] text-neutral-400">
   441→            A09 — Supervision des départs de toutes les compagnies : lecture
   442→            seule, la programmation reste gérée par l’outil de chaque compagnie.
   443→          </p>
   444→        </div>
   445→      </div>
   446→
   447→      {/* ---- Indicateurs du jour ---- */}
   448→      <div className="mt-4 grid grid-cols-2 gap-3 xl:grid-cols-4">
   449→        <Kpi
   450→          index={0}
   451→          label="Départs aujourd’hui"
   452→          value={formatNumber(kpis.count)}
   453→          hint={`${formatNumber(companyDepartures.length)} sur 2 jours`}
   454→          icon={CalendarClock}
   455→          tone="coral"
   456→        />
   457→        <Kpi
   458→          index={1}
   459→          label="Places vendues (jour)"
   460→          value={formatNumber(kpis.sold)}
   461→          hint="toutes compagnies confondues"
   462→          icon={Ticket}
   463→          tone="neutral"
   464→        />
   465→        <Kpi
   466→          index={2}
   467→          label="Taux d’occupation"
   468→          value={`${kpis.occ} %`}
   469→          hint="capacité embarquée aujourd’hui"
   470→          icon={Route}
   471→          tone="coral"
   472→        />
   473→        <Kpi
   474→          index={3}
   475→          label="Départs perturbés"
   476→          value={formatNumber(kpis.perturbes)}
   477→          hint={`${kpis.modifs} modifications suivies`}
   478→          icon={History}
   479→          tone="rose"
   480→        />
   481→      </div>
   482→
   483→      {/* ---- Filtres partagés ---- */}
   484→      <div className="mt-4 flex flex-wrap items-center gap-2.5">
   485→        <div className="relative min-w-[200px] flex-1 sm:max-w-xs">
   486→          <Search className="pointer-events-none absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-neutral-400" />
   487→          <Input
   488→            value={query}
   489→            onChange={(e) => setQuery(e.target.value)}
   490→            placeholder={
   491→              view === "departs"
   492→                ? "Rechercher (trajet, bus, compagnie)…"
   493→                : "Rechercher (trajet, arrêt, compagnie)…"
   494→            }
   495→            aria-label="Rechercher une ligne ou un départ"
   496→            className="h-9 rounded-xl border-neutral-200 bg-white pl-9 text-[13px] focus-visible:ring-coral-200"
   497→          />
   498→        </div>
   499→        {view === "departs" && (
   500→          <>
   501→            <Select value={companyFilter} onValueChange={setCompanyFilter}>
   502→              <SelectTrigger
   503→                aria-label="Filtrer par compagnie"
   504→                className="h-9 w-[170px] rounded-xl border-neutral-200 bg-white text-[13px] shadow-none hover:border-neutral-300"
   505→              >
   506→                <SelectValue />
   507→              </SelectTrigger>
   508→              <SelectContent className="rounded-xl border-neutral-100">
   509→                <SelectItem value="all" className="rounded-lg text-[13px]">
   510→                  Toutes les compagnies
   511→                </SelectItem>
   512→                {companies.map((c) => (
   513→                  <SelectItem
   514→                    key={c.id}
   515→                    value={c.id}
   516→                    className="rounded-lg text-[13px]"
   517→                  >
   518→                    {c.nom}
   519→                  </SelectItem>
   520→                ))}
   521→              </SelectContent>
   522→            </Select>
   523→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   524→              <SelectTrigger
   525→                aria-label="Filtrer par statut"
   526→                className="h-9 w-[140px] rounded-xl border-neutral-200 bg-white text-[13px] shadow-none hover:border-neutral-300"

   535→                  <SelectItem
   536→                    key={key}
   537→                    value={key}
   538→                    className="rounded-lg text-[13px]"
   539→                  >
   540→                    {meta.label}
   541→                  </SelectItem>

   543→              </SelectContent>

   545→            <Select value={dayFilter} onValueChange={setDayFilter}>

   550→                <SelectValue />
   551→              </SelectTrigger>
   552→              <SelectContent className="rounded-xl border-neutral-100">
   553→                <SelectItem value="all" className="rounded-lg text-[13px]">
   554→                  Tous les jours
   555→                </SelectItem>
   556→                <SelectItem value="Aujourd'hui" className="rounded-lg text-[13px]">

  1141→/** Ajoute des minutes à une heure "HH:MM" (utilisée pour l'heure estimée). */
  1142→function shiftTime(heure: string, minutes: number): string {
  1143→  const [h, m] = heure.split(":").map(Number);
  1144→  const total = h * 60 + m + minutes;
  1145→  const hh = Math.floor(total / 60) % 24;
  1146→  const mm = total % 60;
  1147→  return `${String(hh).padStart(2, "0")}:${String(mm).padStart(2, "0")}`;
  1148→}
