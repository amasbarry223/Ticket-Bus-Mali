    77→/** Normalise une chaîne pour la recherche : minuscules, sans espaces/tirets/points. */
    78→function normalizeKey(s: string): string {
    79→  return s
    80→    .toLowerCase()
    81→    .replace(/[\s\-.]/g, "")
    82→    .normalize("NFD")
    83→    .replace(/[\u0300-\u036f]/g, "");
    84→}

    92→/** Étapes de la chaîne critique du point de vue d'une transaction. */
    93→type ChainState = "done" | "pending" | "failed" | "warn";
    94→interface ChainStep {
    95→  label: string;
    96→  note?: string;
    97→  state: ChainState;
    98→}
    99→
   100→function chainFor(t: Transaction): ChainStep[] {
   101→  const anomaly = t.anomalie;
   102→  const resolved = anomaly?.statut === "resolue";
   103→  const emissionFailed = anomaly && !resolved;
   104→
   105→  const last: ChainStep =
   106→    t.statut === "rembourse"
   107→      ? { label: "Émission des billets", note: "Billets annulés — paiement remboursé", state: "warn" }
   108→      : emissionFailed
   109→        ? {
   110→            label: "Émission des billets",
   111→            note: "Débit confirmé mais aucun billet émis — réservation expirée avant le retour opérateur",
   112→            state: "failed",
   113→          }
   114→        : t.statut === "reussi"
   115→          ? {
   116→              label: "Émission des billets",
   117→              note: `${t.billetsEmis} billet${t.billetsEmis > 1 ? "s" : ""} émis après confirmation serveur`,
   118→              state: "done",
   119→            }
   120→          : { label: "Émission des billets", note: "Après confirmation serveur uniquement", state: "pending" };
   121→
   122→  if (t.statut === "en_attente") {
   123→    return [
   124→      { label: "Demande initiée", note: t.date, state: "done" },
   125→      { label: "Requête envoyée à l'opérateur", note: "En attente de validation par le client (USSD)", state: "done" },
   126→      { label: "Débit client", note: "Non encore confirmé", state: "pending" },
   127→      { label: "Confirmation serveur TBM", note: "Après retour fiable de l'opérateur", state: "pending" },
   128→      last,
   129→    ];
   130→  }
   131→  if (t.statut === "echoue") {
   132→    return [
   133→      { label: "Demande initiée", note: t.date, state: "done" },
   134→      { label: "Requête envoyée à l'opérateur", state: "done" },
   135→      { label: "Débit client", note: "Refusé par l'opérateur", state: "failed" },
   136→      { label: "Confirmation serveur TBM", note: "Jamais reçue", state: "pending" },
   137→      last,
   138→    ];
   139→  }
   140→  if (t.statut === "expire") {
   141→    return [
   142→      { label: "Demande initiée", note: t.date, state: "done" },
   143→      { label: "Requête envoyée à l'opérateur", state: "done" },
   144→      { label: "Débit client", note: "Non finalisé par le client", state: "warn" },
   145→      { label: "Confirmation serveur TBM", note: "Places libérées après 10 min (RG1)", state: "warn" },
   146→      last,
   147→    ];
   148→  }
   149→  return [
   150→    { label: "Demande initiée", note: t.date, state: "done" },
   151→    { label: "Requête envoyée à l'opérateur", state: "done" },
   152→    { label: "Débit client", note: `${formatFcfa(t.montant)} — ${t.refOperateur}`, state: "done" },
   153→    { label: "Confirmation serveur TBM", note: "Retour fiable de l'opérateur reçu", state: "done" },
   154→    last,
   155→  ];
   156→}

   232→function CriticalChain({ txn }: { txn: Transaction }) {
   233→  const steps = chainFor(txn);
   234→  return (
   235→    <ol className="space-y-0">
   236→      {steps.map((s, i) => (
   237→        <li key={s.label} className="relative flex gap-3 pb-4 last:pb-0">
   238→          {i < steps.length - 1 && (
   239→            <span
   240→              aria-hidden
   241→              className={cn(
   242→                "absolute left-[9px] top-5 h-full w-0.5",
   243→                s.state === "done" ? "bg-emerald-200" : "bg-neutral-100"
   244→              )}
   245→            />
   246→          )}
   247→          <span
   248→            className={cn(
   249→              "z-10 mt-0.5 flex h-[19px] w-[19px] shrink-0 items-center justify-center rounded-full border-2",
   250→              CHAIN_DOT[s.state]
   251→            )}
   252→          >
   253→            {s.state === "done" && <Check className="h-3 w-3" strokeWidth={3} />}
   254→            {s.state === "failed" && <XCircle className="h-3 w-3" strokeWidth={2.5} />}
   255→            {s.state === "warn" && <AlertTriangle className="h-2.5 w-2.5" strokeWidth={2.5} />}
   256→          </span>
   257→          <div className="min-w-0">
   258→            <p
   259→              className={cn(
   260→                "text-[12.5px] font-semibold",
   261→                s.state === "pending" ? "text-neutral-400" : "text-neutral-700"
   262→              )}
   263→            >
   264→              {s.label}
   265→            </p>
   266→            {s.note && (
   267→              <p
   268→                className={cn(
   269→                  "text-[11.5px] leading-snug",
   270→                  s.state === "failed" ? "text-rose-600" : s.state === "warn" ? "text-amber-600" : "text-neutral-400"
   271→                )}
   272→              >
   273→                {s.note}
   274→              </p>
   275→            )}
   276→          </div>
   277→        </li>
   278→      ))}
   279→    </ol>
   280→  );
   281→}

   410→function ResolveAnomalyDialog({
   411→  txn,
   412→  open,
   413→  onOpenChange,
   414→}: {
   415→  txn: Transaction;
   416→  open: boolean;
   417→  onOpenChange: (v: boolean) => void;
   418→}) {
   419→  const { toast } = useToast();
   420→  const auteur = useAuthStore((s) => s.user?.name) ?? "Opérateur";

   422→  const [note, setNote] = useState("");
   423→  const valid = note.trim().length >= 10;
   424→
   425→  const submit = () => {
   426→    if (!valid) return;
   427→    resolveAnomaly(txn.id, note.trim(), auteur);
   428→    toast({
   429→      title: "Anomalie résolue",
   430→      description: `${txn.id} — la résolution est tracée dans le journal d'audit.`,
   431→    });
   432→    setNote("");
   433→    onOpenChange(false);
   434→  };
   435→
   436→  return (
   437→    <Dialog open={open} onOpenChange={onOpenChange}>
   438→      <DialogContent className="rounded-2xl sm:max-w-[480px]">
   439→        <DialogTitle className="text-[16px] font-bold text-neutral-900">
   440→          Résoudre l&apos;anomalie — {txn.id}
   441→        </DialogTitle>
   442→        <DialogDescription className="text-[12.5px] leading-relaxed text-neutral-500">
   443→          Paiement {formatFcfa(txn.montant)} débité ({txn.refOperateur}) sans billet
   444→          émis. Justifiez la résolution : elle sera horodatée au journal d&apos;audit.
   445→        </DialogDescription>
   446→
   447→        <div className="mt-1 space-y-1.5">
   448→          <Label htmlFor={`note-${txn.id}`} className="text-[12.5px] font-medium text-neutral-700">
   449→            Justification de la résolution *
   450→          </Label>
   451→          <Textarea
   452→            id={`note-${txn.id}`}
   453→            value={note}

   457→          />
   458→          <p className="text-[11px] text-neutral-400">
   459→            Minimum 10 caractères. La transaction passe alors en « anomalie résolue »
   460→            et le rapprochement du jour peut être clôturé.
   461→          </p>
   462→          {txn.id === "TXN-10244" && (
   463→            <p className="rounded-lg bg-violet-50 px-3 py-2 text-[11.5px] leading-relaxed text-violet-700">

   470→
   471→        <div className="mt-2 flex justify-end gap-2">
   472→          <Button
   473→            variant="outline"
   474→            className="h-9 rounded-xl border-neutral-200 text-[13px]"
   475→            onClick={() => onOpenChange(false)}
   476→          >
   477→            Annuler
   478→          </Button>
   479→          <Button
   480→            disabled={!valid}
   481→            className="h-9 gap-1.5 rounded-xl bg-coral-500 px-4 text-[13px] font-semibold text-white hover:bg-coral-600 disabled:opacity-40"
   482→            onClick={submit}
   483→          >
   484→            <ShieldCheck className="h-4 w-4" />
   485→            Enregistrer la résolution
   486→          </Button>
   487→        </div>
   488→      </DialogContent>
   489→    </Dialog>
   490→  );
   491→}
