    73→/** Initiales pour l'avatar-logo d'une compagnie (ex. "Bamako Express" → BE). */
    74→export function companyInitials(nom: string): string {
    75→  return (
    76→    nom
    77→      .split(/\s+/)
    78→      .map((p) => p[0])
    79→      .slice(0, 2)
    80→      .join("")
    81→      .toUpperCase() || "CP"
    82→  );
    83→}
